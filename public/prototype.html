// Architectural Query Factory (TanStack Query v5 & Vertical Slices)
import { queryOptions } from '@tanstack/react-query';
import { api } from '@/lib/api';

export const courseQueries = {
  all: () => ['courses'] as const,

  detail: (id: string) =>
    queryOptions({
      queryKey: [...courseQueries.all(), 'detail', id],
      queryFn: async () => {
        const { data } = await api.get<Course>(`/courses/${id}`);
        return data;
      },
      staleTime: 1000 * 60 * 5,
    }),
};
